1. ESP32 Arduino C++ Web Server Code

Upload this endpoint handler to your ESP32-S3. Ensure CORS headers are enabled so the web dashboard can request telemetry directly:

#include <WiFi.h>
#include <WebServer.h>

WebServer server(80);

void handleDataApi() {
  // CORS Header required for browser access
  server.sendHeader("Access-Control-Allow-Origin", "*");
  
  String json = "{";
  json += "\"pm25\":" + String(readPM25()) + ",";
  json += "\"co_raw\":" + String(analogRead(MQ7_PIN)) + ",";
  json += "\"temperature\":" + String(dht.readTemperature(), 1) + ",";
  json += "\"humidity\":" + String(dht.readHumidity(), 0);
  json += "}";
  
  server.send(200, "application/json", json);
}

void setup() {
  WiFi.begin("YOUR_WIFI_SSID", "YOUR_WIFI_PASSWORD");
  server.on("/api/data", handleDataApi);
  server.begin();
}

void loop() { server.handleClient(); }
